cmake_minimum_required(VERSION 3.16)
project(idz3 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_definitions(_POSIX_C_SOURCE=200809L)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)

set(COMMON common/config.c common/shared_sim.c)
set(MQ_COMMON independent_mq/mq_common.c)

add_executable(parent_named parent_named/main.c ${COMMON})
target_link_libraries(parent_named PRIVATE rt Threads::Threads)

add_executable(parent_unnamed parent_unnamed/main.c ${COMMON})
target_compile_definitions(parent_unnamed PRIVATE USE_UNNAMED_SEMAPHORES=1)
target_link_libraries(parent_unnamed PRIVATE rt Threads::Threads)

foreach(tool shm_init shm_reader shm_writer shm_stop shm_status shm_cleanup)
    add_executable(${tool} independent_shm/tool.c ${COMMON})
    target_link_libraries(${tool} PRIVATE rt Threads::Threads)
endforeach()

add_executable(mq_server independent_mq/server.c ${MQ_COMMON} ${COMMON})
target_link_libraries(mq_server PRIVATE rt Threads::Threads)
add_executable(mq_reader independent_mq/reader.c ${MQ_COMMON})
target_link_libraries(mq_reader PRIVATE rt Threads::Threads)
add_executable(mq_writer independent_mq/writer.c ${MQ_COMMON})
target_link_libraries(mq_writer PRIVATE rt Threads::Threads)
foreach(tool mq_stop mq_cleanup)
    add_executable(${tool} independent_mq/control.c)
    target_link_libraries(${tool} PRIVATE rt Threads::Threads)
endforeach()
